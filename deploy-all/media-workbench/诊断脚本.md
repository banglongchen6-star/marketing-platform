# Media Workbench 诊断脚本

请在浏览器中打开 `index.html`，然后按 F12 打开开发者工具，在 **Console** 标签中粘贴并运行以下脚本：

## 诊断脚本 1：检查按钮状态

```javascript
// 检查按钮是否被正确定义
console.log('=== 按钮检查 ===');
console.log('TOPBAR_ACTIONS:', TOPBAR_ACTIONS);
console.log('TOPBAR_ACTIONS keys:', Object.keys(TOPBAR_ACTIONS || {}));

// 检查关键函数是否存在
console.log('\n=== 函数检查 ===');
console.log('openSampleModal:', typeof openSampleModal);
console.log('openModal:', typeof openModal);
console.log('closeModal:', typeof closeModal);
console.log('navigate:', typeof navigate);

// 检查模态框元素
console.log('\n=== 模态框状态 ===');
['sample-modal', 'material-modal', 'settlement-modal', 'candidate-modal', 'content-modal', 'schedule-modal'].forEach(id => {
  const el = document.getElementById(id);
  if (el) {
    console.log(`${id}: display=${getComputedStyle(el).display}, z-index=${getComputedStyle(el).zIndex}`);
  } else {
    console.log(`${id}: NOT FOUND`);
  }
});

// 检查是否有遮挡层
console.log('\n=== 遮挡层检查 ===');
const overlays = document.querySelectorAll('.modal-overlay');
overlays.forEach(el => {
  const style = getComputedStyle(el);
  if (style.display !== 'none') {
    console.log('⚠️ 发现可见遮挡层:', el.id, 'display=', style.display);
  }
});

console.log('\n✅ 检查完成');
```

## 诊断脚本 2：强制打开模态框

```javascript
// 强制打开样品模态框
console.log('尝试打开样品模态框...');
try {
  openSampleModal();
  console.log('✅ openSampleModal() 调用成功');
} catch (e) {
  console.log('❌ 错误:', e.message);
}

// 检查模态框是否显示
const modal = document.getElementById('sample-modal');
if (modal) {
  console.log('样品模态框 display:', getComputedStyle(modal).display);
}
```

## 诊断脚本 3：检查 CSS 布局

```javascript
// 检查统计卡片布局
console.log('=== 统计卡片布局 ===');
const statsGrid = document.querySelector('.stats-grid');
if (statsGrid) {
  const style = getComputedStyle(statsGrid);
  console.log('grid-template-columns:', style.gridTemplateColumns);
  console.log('display:', style.display);
  console.log('gap:', style.gap);
} else {
  console.log('❌ 未找到 .stats-grid 元素');
}

// 检查统计卡片数量
const statCards = document.querySelectorAll('.stat-card');
console.log('统计卡片数量:', statCards.length);
```

## 诊断脚本 4：一键测试所有功能

```javascript
(function() {
  console.log('\n========================================');
  console.log('Media Workbench 完整诊断');
  console.log('========================================\n');
  
  // 1. 检查全局对象
  console.log('【1】全局对象检查');
  const globals = ['TOPBAR_ACTIONS', 'DB', 'DEFAULTS', 'currentUser'];
  globals.forEach(name => {
    console.log(`  ${name}: ${typeof window[name] !== 'undefined' ? '✅ 存在' : '❌ 不存在'}`);
  });
  
  // 2. 检查关键函数
  console.log('\n【2】关键函数检查');
  const functions = ['openSampleModal', 'openMaterialModal', 'openSettlementModal', 
                     'openCandidateModal', 'openContentModal', 'openModal', 'closeModal', 'navigate'];
  functions.forEach(name => {
    console.log(`  ${name}: ${typeof window[name] === 'function' ? '✅ 可用' : '❌ 不可用'}`);
  });
  
  // 3. 检查模态框元素
  console.log('\n【3】模态框元素检查');
  const modals = ['sample-modal', 'material-modal', 'settlement-modal', 
                  'candidate-modal', 'content-modal', 'schedule-modal'];
  modals.forEach(id => {
    const el = document.getElementById(id);
    if (el) {
      const display = getComputedStyle(el).display;
      console.log(`  ${id}: ${display === 'none' ? '✅ 隐藏' : '⚠️ 显示(' + display + ')'}`);
    } else {
      console.log(`  ${id}: ❌ 不存在`);
    }
  });
  
  // 4. 检查按钮元素
  console.log('\n【4】顶部按钮检查');
  const buttons = document.querySelectorAll('.topbar-actions .btn');
  console.log(`  找到 ${buttons.length} 个按钮`);
  buttons.forEach((btn, i) => {
    console.log(`  按钮${i+1}: "${btn.textContent.trim()}" - ${btn.onclick ? '✅ 有onclick' : '❌ 无onclick'}`);
  });
  
  // 5. 检查统计卡片
  console.log('\n【5】统计卡片布局');
  const statsGrid = document.querySelector('.stats-grid');
  if (statsGrid) {
    const style = getComputedStyle(statsGrid);
    console.log(`  grid-template-columns: ${style.gridTemplateColumns}`);
    console.log(`  卡片数量: ${document.querySelectorAll('.stat-card').length}`);
  }
  
  // 6. 检查是否有 JS 错误
  console.log('\n【6】尝试调用 openSampleModal()');
  try {
    openSampleModal();
    console.log('  ✅ 调用成功');
    // 关闭它
    setTimeout(() => {
      closeModal('sample-modal');
      console.log('  ✅ 已关闭');
    }, 500);
  } catch (e) {
    console.log('  ❌ 错误:', e.message);
  }
  
  console.log('\n========================================');
  console.log('诊断完成！');
  console.log('========================================\n');
})();
```

## 常见问题

### Q: 控制台显示 "TOPBAR_ACTIONS is not defined"
A: 说明 JavaScript 文件没有正确加载，可能是浏览器缓存问题。

### Q: 按钮存在但 onclick 为 null
A: 说明 HTML 加载了但 JavaScript 没有执行，检查控制台是否有红色错误。

### Q: 模态框已经显示
A: 说明某个模态框被意外打开了，刷新页面试试。

### Q: grid-template-columns 显示 2 列
A: 可能是窗口宽度小于 480px，或者 CSS 没有正确加载。
